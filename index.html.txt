        const planData = generatePlan();
        let state = JSON.parse(localStorage.getItem('90dias_fe_v4')) || {};
        let currentMonth = 1;
        let currentWeek = 1;
        let chartInstance = null;

        const saveProgress = (dayId, task, value) => {
            if (!state[dayId]) state[dayId] = {};
            state[dayId][task] = value;
            localStorage.setItem('90dias_fe_v4', JSON.stringify(state));
            updateStats();
            refreshCurrentView();
            showToast();
        };

        const showToast = () => {
            const t = document.getElementById('toast');
            t.style.opacity = '1';
            setTimeout(() => { t.style.opacity = '0'; }, 1500);
        };

        const updateStats = () => {
            let completed = 0, fullDays = 0, bibleCaps = 0, prayers = 0;
            Object.keys(state).forEach(id => {
                const day = state[id];
                let count = 0;
                if(day.bible) { completed++; bibleCaps += 3; count++; }
                if(day.egw) { completed++; count++; }
                if(day.pm) { completed++; prayers++; count++; }
                if(day.pt) { completed++; prayers++; count++; }
                if(count === 4) fullDays++;
            });
            const percent = Math.round((completed / (90 * 4)) * 100);
            document.getElementById('total-percent').innerText = `${percent}%`;
            document.getElementById('total-bar').style.width = `${percent}%`;
            document.getElementById('stat-days').innerText = `${fullDays}/90`;
            document.getElementById('stat-bible').innerText = bibleCaps;
            document.getElementById('stat-prayers').innerText = prayers;
            document.getElementById('stat-fasts').innerText = `${Math.floor(fullDays/15)}/6`;
            renderChart(percent);
        };

        const renderChart = (percent) => {
            const ctx = document.getElementById('progressChart').getContext('2d');
            if(chartInstance) chartInstance.destroy();
            chartInstance = new Chart(ctx, {
                type: 'doughnut',
                data: { datasets: [{ data: [percent, 100 - percent], backgroundColor: ['#007AFF', '#E5E5EA'], borderWidth: 0 }] },
                options: { cutout: '82%', responsive: true, maintainAspectRatio: false, plugins: { tooltip: { enabled: false } } }
            });
        };

        const selectMonth = (id) => {
            currentMonth = parseInt(id.replace('mes', ''));
            const titles = ["", "Mes 1: Evangelios", "Mes 2: Hechos y Pablo", "Mes 3: Sabiduría"];
            document.getElementById('current-month-title').innerText = titles[currentMonth];
            document.querySelectorAll('nav button').forEach(b => b.classList.remove('bg-white', 'shadow-md', 'text-ios-blue'));
            document.getElementById(`nav-mes${currentMonth}`).classList.add('bg-white', 'shadow-md', 'text-ios-blue');
            renderWeeks();
            selectWeek((currentMonth - 1) * 4 + 1);
        };

        const renderWeeks = () => {
            const container = document.getElementById('week-tabs');
            container.innerHTML = '';
            const start = (currentMonth - 1) * 4 + 1;
            for(let i=start; i < Math.min(start+4, 14); i++) {
                const btn = document.createElement('button');
                btn.className = `px-4 py-1.5 rounded-full text-xs font-bold transition-all ${currentWeek === i ? 'bg-white shadow-sm text-ios-blue' : 'text-ios-secondary'}`;
                btn.innerText = `S${i}`;
                btn.onclick = () => selectWeek(i);
                container.appendChild(btn);
            }
        };

        const selectWeek = (num) => {
            currentWeek = num;
            renderWeeks();
            const weekDays = planData.filter(d => d.week === num);
            const grid = document.getElementById('days-grid');
            grid.innerHTML = '';
            weekDays.forEach(day => {
                const dayState = state[day.id] || {};
                const completedCount = Object.values(dayState).filter(v => v).length;
                const isAyuno = day.isFasting;
                const card = document.createElement('div');
                card.className = `day-card glass p-6 rounded-ios-lg cursor-pointer flex flex-col gap-4 border-2 ${isAyuno ? 'border-ios-gold/20 bg-ios-gold/5' : 'border-transparent'}`;

'✨ Ayuno Especial' : 'Plan Diario'}</h4></div><div class="w-8 h-8 rounded-full border-2 border-gray-100 flex items-center justify-center text-[10px] font-bold ${completedCount === 4 ? 'bg-ios-green border-ios-green text-white' : ''}">${completedCount}/4</div></div><div class="space-y-2"><div class="flex items-start gap-2"><span class="text-sm">📖</span><div class="flex flex-col"><span class="text-xs font-bold text-ios-text">${day.bible}</span><span class="text-[10px] text-ios-secondary uppercase">Biblia</span></div></div><div class="flex items-start gap-2"><span class="text-sm">📚</span><div class="flex flex-col"><span class="text-xs font-bold text-ios-text line-clamp-1">${day.egw}</span><span class="text-[10px] text-ios-secondary uppercase">EGW</span></div></div></div>`;
                grid.appendChild(card);
            });
        };

        const refreshCurrentView = () => selectWeek(currentWeek);

        const openModal = (dayId) => {
            const day = planData.find(d => d.id === dayId);
            const dayState = state[dayId] || {};

            renderModalCheck('modal-check-pt', dayId, 'pt', 'Oración Tarde', dayState.pt);

        const renderModalCheck = (containerId, dayId, task, label, checked) => {
            document.getElementById(containerId).innerHTML = `<label class="flex items-center gap-4 cursor-pointer"><input type="checkbox" class="ios-checkbox" ${checked ? 'checked' : ''} onchange="saveProgress('${dayId}', '${task}', this.checked)"><div class="flex flex-col"><span class="text-sm font-bold ${checked ? 'text-ios-secondary line-through' : ''}">${label}</span><span class="text-[10px] text-ios-secondary uppercase tracking-widest">${task}</span></div></label>`;
